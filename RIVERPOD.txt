import 'package:flutter/material.dart';
import 'package:flutter_riverpod/flutter_riverpod.dart';

void main() {
  runApp(
    ProviderScope(
      child: MyApp(),
    ),
  );
}

class MyApp extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      home: BookPage(),
      debugShowCheckedModeBanner: false,
    );
  }
}

// ---------------- MODEL ----------------
class Book {
  String title;
  String author;
  int price;
  int qty;

  Book(this.title, this.author, this.price, this.qty);
}

// ---------------- PROVIDER ----------------
final cartProvider = StateProvider<List<Book>>((ref) => []);

// ---------------- UI ----------------
class BookPage extends ConsumerWidget {
  final List<Book> books = [
    Book("Flutter Guide", "John", 500, 0),
    Book("Dart Basics", "Alice", 400, 0),
    Book("UI Design", "Mark", 300, 0),
  ];

  @override
  Widget build(BuildContext context, WidgetRef ref) {
    final cart = ref.watch(cartProvider);

    int total = 0;
    for (var b in cart) {
      total += b.price * b.qty;
    }

    return Scaffold(
      appBar: AppBar(
        title: Text("Book Store"),
      ),
      body: Column(
        children: [

          // ---------------- BOOK LIST ----------------
          Expanded(
            child: ListView(
              children: books.map((book) {
                return ListTile(
                  title: Text(book.title),
                  subtitle: Text("${book.author} - ₹${book.price}"),
                  trailing: ElevatedButton(
                    child: Text("Add"),
                    onPressed: () {
                      bool found = false;
                      List<Book> updated = [...cart];

                      for (var item in updated) {
                        if (item.title == book.title) {
                          item.qty++;
                          found = true;
                        }
                      }

                      if (!found) {
                        updated.add(
                          Book(
                            book.title,
                            book.author,
                            book.price,
                            1,
                          ),
                        );
                      }

                      ref.read(cartProvider.notifier).state = updated;
                    },
                  ),
                );
              }).toList(),
            ),
          ),

          Divider(),

          // ---------------- CART TITLE ----------------
          Padding(
            padding: EdgeInsets.all(8),
            child: Text(
              "Cart",
              style: TextStyle(
                fontSize: 20,
                fontWeight: FontWeight.bold,
              ),
            ),
          ),

          // ---------------- CART LIST ----------------
          Expanded(
            child: ListView(
              children: cart.map((book) {
                return ListTile(
                  title: Text(book.title),
                  subtitle: Text("Qty: ${book.qty}"),
                  trailing: Row(
                    mainAxisSize: MainAxisSize.min,
                    children: [

                      // Decrease
                      IconButton(
                        icon: Icon(Icons.remove),
                        onPressed: () {
                          book.qty--;

                          List<Book> updated = [...cart];
                          updated.removeWhere((b) => b.qty <= 0);

                          ref.read(cartProvider.notifier).state = updated;
                        },
                      ),

                      // Increase
                      IconButton(
                        icon: Icon(Icons.add),
                        onPressed: () {
                          book.qty++;

                          ref.read(cartProvider.notifier).state = [...cart];
                        },
                      ),
                    ],
                  ),
                );
              }).toList(),
            ),
          ),

          // ---------------- TOTAL ----------------
          Padding(
            padding: EdgeInsets.all(15),
            child: Text(
              "Total: ₹$total",
              style: TextStyle(
                fontSize: 22,
                fontWeight: FontWeight.bold,
              ),
            ),
          ),
        ],
      ),
    );
  }
}
